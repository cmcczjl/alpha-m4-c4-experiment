#!/usr/bin/env python3
import os
import sys
import json
import requests

def main() -> None:
    # GitHub token for higher rate limits; optional
    token = os.getenv("GITHUB_TOKEN")
    headers = {}
    if token:
        headers["Authorization"] = f"token {token}"

    # Search query: bounty or help wanted labels, keywords n8n or automation
    query = 'label:"bounty" OR label:"help wanted" n8n OR automation'
    url = f"https://api.github.com/search/issues?q={query}&per_page=100"

    resp = requests.get(url, headers=headers)
    if resp.status_code != 200:
        print(f"GitHub API error: {resp.status_code}", file=sys.stderr)
        sys.exit(1)

    data = resp.json()
    items = data.get("items", [])[:3]  # take first 3 results

    leads = []
    for item in items:
        leads.append(
            {
                "url": item["html_url"],
                "title": item["title"],
                "description": item["body"][:200],  # truncate for brevity
            }
        )

    os.makedirs("data", exist_ok=True)
    with open("data/market_leads.json", "w", encoding="utf-8") as f:
        json.dump(leads, f, indent=2)

    print(f"Saved {len(leads)} leads to data/market_leads.json")

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Check GitHub CLI authentication status and log the result.
"""
import subprocess
import datetime
import os

def main():
    # Run `gh auth status` command
    try:
        result = subprocess.run(['gh', 'auth', 'status'], capture_output=True, text=True, check=True)
        output = result.stdout.strip()
        error = result.stderr.strip()
    except subprocess.CalledProcessError as e:
        output = e.stdout.strip() if e.stdout else ''
        error = e.stderr.strip() if e.stderr else ''
        output += f"\nError: {e}"
    except FileNotFoundError:
        output = ""
        error = "GitHub CLI (gh) not found. Please install it."

    timestamp = datetime.datetime.utcnow().isoformat()
    log_entry = f"{timestamp} | output:\n{output}\nerror:\n{error}\n{'-'*80}\n"

    # Ensure data directory exists
    os.makedirs('data', exist_ok=True)
    log_path = os.path.join('data', 'gh_auth.log')

    # Append log entry to file
    with open(log_path, 'a', encoding='utf-8') as f:
        f.write(log_entry)

    # Print to console for immediate visibility
    print(log_entry)

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
import zipfile
import hashlib
import sys
import os

def compute_md5(data: bytes) -> str:
    return hashlib.md5(data).hexdigest()

def main(zip_path: str):
    if not os.path.isfile(zip_path):
        print(f"Error: '{zip_path}' not found.", file=sys.stderr)
        sys.exit(1)

    try:
        with zipfile.ZipFile(zip_path, 'r') as zf:
            for info in zf.infolist():
                # Skip directories
                if info.is_dir():
                    continue
                file_data = zf.read(info.filename)
                md5_hash = compute_md5(file_data)
                print(f"{info.filename}: {md5_hash}")
    except zipfile.BadZipFile:
        print(f"Error: '{zip_path}' is not a valid zip file.", file=sys.stderr)
        sys.exit(1)

def gh_search_issues(query, per_page=30, max_pages=2):
    """Search GitHub issues via REST API with pagination."""
    encoded = urllib.parse.quote(query)
    results = []
    for page in range(1, max_pages + 1):
        url = f'https://api.github.com/search/issues?q={encoded}&sort=created&order=desc&per_page={per_page}&page={page}'
        req = urllib.request.Request(url)
        req.add_header('Accept', 'application/vnd.github.v3+json')
        req.add_header('User-Agent', 'market-research-bot')
        if GITHUB_TOKEN:
            req.add_header('Authorization', f'token {GITHUB_TOKEN}')
        try:
            with urllib.request.urlopen(req, timeout=30) as resp:
                data = json.loads(resp.read().decode())
                items = data.get('items', [])
                results.extend(items)
                if len(items) < per_page:
                    break
        except Exception as e:
            print(f'  [WARN] page {page} failed: {e}')
            break
    return results
