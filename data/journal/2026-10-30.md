    """
    Very simple static analysis:
    - Syntax check (via ast.parse)
    - Detect functions without docstrings
    - Detect overly long lines (> 120 chars)
    Returns a multi‑line review string.
    """
    review_lines = []
    try:
        tree = ast.parse(source)
    except SyntaxError as e:
        return f"❌ Syntax error: {e.msg} (line {e.lineno})"

    # Check for functions without docstrings
    for node in ast.walk(tree):
        if isinstance(node, ast.FunctionDef):

def get_cpu():

def get_mem():
    try:
        import psutil
        return psutil.virtual_memory().total
    except Exception:
        # fallback using os.sysconf
        pages = os.sysconf('SC_PHYS_PAGES')
        page_size = os.sysconf('SC_PAGE_SIZE')
        return pages * page_size

def main():

#!/usr/bin/env python3
import os
import sys

# 默认的白名单文件路径
WHITELIST_PATH = "/etc/security/whitelist.conf"

def parse_whitelist(content: str):
    """
    简单解析白名单文件，返回允许的命令集合。
    假设每行一个命令，忽略空行和以 # 开头的注释。
    """
    commands = []
    for line in content.splitlines():
        line = line.strip()
        if not line or line.startswith('#'):
            continue
        # 支持以空格或逗号分隔的多命令写法
        for part in line.replace(',', ' ').split():
            part = part.strip()
            if part:
                commands.append(part)
    return commands

def main():
    if not os.path.isfile(WHITELIST_PATH):
        print(f"白名单文件未找到: {WHITELIST_PATH}", file=sys.stderr)
        sys.exit(1)

    try:
        with open(WHITELIST_PATH, "r", encoding="utf-8") as f:
            raw = f.read()
    except PermissionError:
        print(f"没有读取白名单文件的权限: {WHITELIST_PATH}", file=sys.stderr)
        sys.exit(1)

    allowed_cmds = parse_whitelist(raw)

    if not allowed_cmds:
        print("白名单中未检测到任何允许的命令。")
    else:
        print("当前允许的命令集合:")
        for cmd in sorted(set(allowed_cmds)):
            print(f"- {cmd}")

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
import os
import shutil
import sys

def main():
    # Determine if running as root
    is_root = os.getuid() == 0

    # Check for presence of commands
    has_sudo = shutil.which('sudo') is not None
    has_apt = shutil.which('apt') is not None
    has_apt_get = shutil.which('apt-get') is not None

    cmd = None

    if is_root:
        if has_apt:
            cmd = 'apt'
        elif has_apt_get:
            cmd = 'apt-get'
    else:
        if has_sudo:
            if has_apt:
                cmd = 'sudo apt'
            elif has_apt_get:
                cmd = 'sudo apt-get'

    if cmd:
        print(f"使用 {cmd} …")
    else:
        print("需手动安装或使用其他方式")

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
import subprocess
import sys

def run_cmd(cmd):
    """Run a shell command and return its stdout, stderr, and exit code."""
    try:
        result = subprocess.run(
            cmd,
            shell=True,
            capture_output=True,
            text=True,
            check=False  # We will handle non-zero exit codes ourselves
        )
        return result.stdout.strip(), result.stderr.strip(), result.returncode
    except Exception as e:
        return "", str(e), -1

def main():
    commands = [
        ("which act", "act location"),
        ("docker ps", "docker ps output"),
        ("act --list", "act list output")
    ]

    for cmd, description in commands:
        print(f"\n=== Running: {cmd} ===")
        out, err, rc = run_cmd(cmd)
        if rc == 0:
            print(f"[SUCCESS] {description}:\n{out}")
        else:
            print(f"[ERROR] {description} returned non-zero exit code {rc}")
            if out:
                print(f"stdout:\n{out}")
            if err:
                print(f"stderr:\n{err}")

if __name__ == "__main__":
    sys.exit(main())
